                          <label className="block text-xs font-bold text-slate-500 uppercase mb-1">Service</label>
                          <select
                            value={newPatient.service}
                            onChange={(e) => setNewPatient({ ...newPatient, service: e.target.value })}
                            className="w-full px-4 py-2.5 bg-slate-50 border border-slate-200 rounded-xl text-sm outline-none focus:border-blue-500 cursor-pointer">
                            <option value="Médecine Générale">Médecine Générale</option>
                            <option value="Maternité (CPN)">Maternité (CPN)</option>
                            <option value="Urgences">Urgences</option>
                            <option value="Laboratoire">Laboratoire</option>
                          </select>
                        </div>
                        <div>
                          <label className="block text-xs font-bold text-slate-500 uppercase mb-1">Médecin Référent</label>
                          <select
                            value={newPatient.medecin}
                            onChange={(e) => setNewPatient({ ...newPatient, medecin: e.target.value })}
                            className="w-full px-4 py-2.5 bg-slate-50 border border-slate-200 rounded-xl text-sm outline-none focus:border-blue-500 cursor-pointer">
                            <option value="Dr. Ibrahima">Dr. Ibrahima</option>
                            <option value="Dr. Mariam Sidibé">Dr. Mariam Sidibé</option>
                            <option value="Dr. Alou Diallo">Dr. Alou Diallo</option>
                          </select>
                        </div>
                      </div>

                      <div className="pt-4 flex items-center justify-end gap-3 border-t border-slate-100">
                        <button
                          type="button"
                          onClick={() => setIsModalOpen(false)}
                          className="px-4 py-2.5 bg-slate-100 hover:bg-slate-200 text-slate-700 rounded-xl text-sm font-semibold transition-colors cursor-pointer">
                          Annuler
                        </button>
                        <button
                          type="submit"
                          className="px-5 py-2.5 bg-blue-600 hover:bg-blue-700 text-white rounded-xl text-sm font-semibold transition-all shadow-xs cursor-pointer">
                          Valider l'admission
                        </button>
                      </div>
                    </form>
                  </div>
                </div>
              )}
            </div>
          )}

          {/* Affichage pour les autres modules en cours de développement */}
          {activeTab !== "dashboard" && activeTab !== "kpis" && activeTab !== "admissions" && (
            <div className="bg-white p-12 rounded-2xl border border-slate-200/80 text-center space-y-4 shadow-xs">
              <div className="w-16 h-16 bg-blue-50 text-blue-600 rounded-2xl flex items-center justify-center mx-auto">
                <CheckCircle2 size={32} />
              </div>
              <h3 className="text-xl font-bold text-slate-800 tracking-tight">
                Module : {getPageTitle()}
              </h3>
              <p className="text-sm text-slate-500 max-w-md mx-auto">
                Le module Admissions & Patients est désormais pleinement implémenté ! C'est au tour du prochain module de notre liste : **Prise de Rendez-vous**.
              </p>
              <div className="pt-2 flex items-center justify-center gap-3">
                <button
                  onClick={() => setActiveTab("dashboard")}
                  className="bg-slate-100 hover:bg-slate-200 text-slate-700 px-5 py-2.5 rounded-xl text-sm font-semibold transition-colors cursor-pointer">
                  Retour au tableau de bord
                </button>
              </div>
            </div>
          )}
        </div>
      </main>
    </div>
  );
